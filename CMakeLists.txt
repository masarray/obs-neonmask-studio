cmake_minimum_required(VERSION 3.20)
project(obs-neonmask-studio VERSION 0.1.0 LANGUAGES C)

option(NEONMASK_BUILD_PLUGIN "Build the native OBS filter" ON)
option(NEONMASK_BUILD_TESTS "Build headless geometry tests" ON)
option(NEONMASK_BUILD_SHADER_SMOKE "Compile effect on a live libobs graphics device" OFF)

add_library(neonmask-math STATIC src/neonmask-math.c src/neonmask-presets.c src/neonmask-config.c src/neonmask-motion.c src/neonmask-art.c src/neonmask-shapes.c src/neonmask-svg.c src/neonmask-safe-fit.c)
target_include_directories(neonmask-math PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src")
target_compile_features(neonmask-math PUBLIC c_std_11)
# The art/perimeter helper adds atan2f/fmodf to the shared static library.
# Link libm transitively for every headless test and native plugin consumer.
if(UNIX AND NOT APPLE)
  target_link_libraries(neonmask-math PUBLIC m)
endif()
set_target_properties(neonmask-math PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(NOT MSVC)
  target_compile_options(neonmask-math PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(NEONMASK_BUILD_PLUGIN)
  find_package(libobs CONFIG REQUIRED)
  add_library(obs-neonmask MODULE src/plugin-main.c src/neonmask-filter.c)
  target_include_directories(obs-neonmask PRIVATE src)
  target_compile_features(obs-neonmask PRIVATE c_std_11)
  configure_file(src/neonmask-version.h.in "${CMAKE_CURRENT_BINARY_DIR}/generated/neonmask-version.h" @ONLY)
  target_include_directories(obs-neonmask PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
  target_link_libraries(obs-neonmask PRIVATE OBS::libobs neonmask-math)
  if(UNIX AND NOT APPLE)
    target_link_libraries(obs-neonmask PRIVATE m)
  endif()
  if(NOT MSVC)
    target_compile_options(obs-neonmask PRIVATE -Wall -Wextra -Wpedantic)
  endif()
  set_target_properties(obs-neonmask PROPERTIES PREFIX "" OUTPUT_NAME "obs-neonmask")
  if(WIN32)
    set(NEONMASK_PLUGIN_DIR "obs-plugins/64bit")
    set(NEONMASK_DATA_DIR "data/obs-plugins/obs-neonmask")
  elseif(APPLE)
    set(NEONMASK_PLUGIN_DIR "lib/obs-plugins")
    set(NEONMASK_DATA_DIR "share/obs/obs-plugins/obs-neonmask")
  else()
    include(GNUInstallDirs)
    set(NEONMASK_PLUGIN_DIR "${CMAKE_INSTALL_LIBDIR}/obs-plugins")
    set(NEONMASK_DATA_DIR "${CMAKE_INSTALL_DATADIR}/obs/obs-plugins/obs-neonmask")
  endif()
  install(TARGETS obs-neonmask LIBRARY DESTINATION "${NEONMASK_PLUGIN_DIR}" RUNTIME DESTINATION "${NEONMASK_PLUGIN_DIR}")
  install(DIRECTORY shaders/ DESTINATION "${NEONMASK_DATA_DIR}/shaders")
  install(DIRECTORY data/locale/ DESTINATION "${NEONMASK_DATA_DIR}/locale")
  # Original local-only shape examples can be browsed from the SVG property.
  install(DIRECTORY assets/designs/ DESTINATION "${NEONMASK_DATA_DIR}/designs"
          FILES_MATCHING PATTERN "*.svg")
endif()

if(NEONMASK_BUILD_TESTS)
  enable_testing()
  add_executable(neonmask-math-tests tests/test-math.c)
  target_link_libraries(neonmask-math-tests PRIVATE neonmask-math)
  if(NOT MSVC)
    target_link_libraries(neonmask-math-tests PRIVATE m)
  endif()
  add_test(NAME neonmask-math-tests COMMAND neonmask-math-tests)
  add_executable(neonmask-preset-tests tests/test-presets.c)
  target_link_libraries(neonmask-preset-tests PRIVATE neonmask-math)
  add_test(NAME neonmask-preset-tests COMMAND neonmask-preset-tests)
  add_executable(neonmask-config-tests tests/test-config.c)
  target_link_libraries(neonmask-config-tests PRIVATE neonmask-math)
  if(NOT MSVC)
    target_link_libraries(neonmask-config-tests PRIVATE m)
  endif()
  add_test(NAME neonmask-config-tests COMMAND neonmask-config-tests)
  add_executable(neonmask-motion-tests tests/test-motion.c)
  target_link_libraries(neonmask-motion-tests PRIVATE neonmask-math)
  if(NOT MSVC)
    target_link_libraries(neonmask-motion-tests PRIVATE m)
  endif()
  add_test(NAME neonmask-motion-tests COMMAND neonmask-motion-tests)
  add_executable(neonmask-art-tests tests/test-art.c)
  target_link_libraries(neonmask-art-tests PRIVATE neonmask-math)
  if(NOT MSVC)
    target_link_libraries(neonmask-art-tests PRIVATE m)
  endif()
  add_test(NAME neonmask-art-tests COMMAND neonmask-art-tests)
  add_executable(neonmask-authored-shapes-tests tests/test-authored-shapes.c)
  target_link_libraries(neonmask-authored-shapes-tests PRIVATE neonmask-math)
  add_test(NAME neonmask-authored-shapes-tests COMMAND neonmask-authored-shapes-tests)
  add_executable(neonmask-safe-fit-tests tests/test-safe-fit.c)
  target_link_libraries(neonmask-safe-fit-tests PRIVATE neonmask-math)
  add_test(NAME neonmask-safe-fit-tests COMMAND neonmask-safe-fit-tests)
  add_executable(neonmask-svg-tests tests/test-svg.c)
  target_link_libraries(neonmask-svg-tests PRIVATE neonmask-math)
  add_test(NAME neonmask-svg-tests COMMAND neonmask-svg-tests
           "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/d24-local-mask.svg"
           "${CMAKE_CURRENT_SOURCE_DIR}/assets/designs/chat-bubble.svg"
           "${CMAKE_CURRENT_SOURCE_DIR}/assets/designs/angled-card.svg")
  if(NEONMASK_BUILD_PLUGIN AND NEONMASK_BUILD_SHADER_SMOKE)
    add_executable(neonmask-gpu-smoke tests/test-gpu-smoke.c)
    target_link_libraries(neonmask-gpu-smoke PRIVATE OBS::libobs)
    if(UNIX AND NOT APPLE)
      target_link_libraries(neonmask-gpu-smoke PRIVATE m)
    endif()
    add_test(NAME neonmask-gpu-smoke
             COMMAND neonmask-gpu-smoke "${CMAKE_CURRENT_SOURCE_DIR}/shaders/neon-mask.effect")
    set_tests_properties(neonmask-gpu-smoke PROPERTIES TIMEOUT 60)
  endif()
  find_package(Python3 COMPONENTS Interpreter QUIET)
  if(Python3_Interpreter_FOUND)
    add_test(NAME neonmask-shader-contract COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_shader_contract.py")
  endif()
endif()
